C) Python tester test_api.py

import argparse, sys, time, json
import yaml
import requests
from requests.adapters import HTTPAdapter
from urllib3.util.retry import Retry

def load_yaml(p):
    with open(p, "r", encoding="utf-8") as f:
        return yaml.safe_load(f)

def session_with_retries(total=3, backoff=0.3):
    s = requests.Session()
    retry = Retry(
        total=total,
        connect=total,
        read=total,
        status=total,
        backoff_factor=backoff,
        status_forcelist=(429, 500, 502, 503, 504),
        allowed_methods=frozenset(["GET","POST","PUT","PATCH","DELETE"])
    )
    adapter = HTTPAdapter(max_retries=retry)
    s.mount("https://", adapter)
    s.mount("http://", adapter)
    return s

def get_token(cfg, verify, ca_bundle, timeout):
    auth = cfg.get("auth", {})
    token_url = auth.get("token_url")
    if not token_url:
        return None  # assume no token needed
    data = {"grant_type": "client_credentials"}
    if auth.get("scope"): data["scope"] = auth["scope"]
    if auth.get("audience"): data["audience"] = auth["audience"]
    resp = requests.post(
        token_url,
        data=data,
        auth=(auth.get("client_id",""), auth.get("client_secret","")),
        timeout=timeout,
        verify=ca_bundle if ca_bundle else verify
    )
    resp.raise_for_status()
    tok = resp.json().get("access_token")
    if not tok:
        raise RuntimeError("Token fetch succeeded but access_token missing in response")
    return tok

def check_json_keys(obj, keys):
    if not isinstance(obj, dict): return False, "Response is not a JSON object"
    missing = [k for k in keys or [] if k not in obj]
    return (len(missing) == 0, f"Missing keys: {missing}" if missing else "ok")

def main():
    parser = argparse.ArgumentParser(description="TLS-verified API tester")
    parser.add_argument("--config", default="config.yaml")
    parser.add_argument("--timeout", type=float, default=10.0)
    parser.add_argument("--ca-bundle", default=None, help="Path to custom CA bundle (PEM).")
    parser.add_argument("--mtls-cert", default=None, help="Path to client cert (PEM).")
    parser.add_argument("--mtls-key", default=None, help="Path to client key (PEM).")
    args = parser.parse_args()

    cfg = load_yaml(args.config)
    base = cfg["api"]["base_url"].rstrip("/")
    default_headers = cfg["api"].get("default_headers", {}) or {}

    # Strict TLS: verify must be True unless a bundle is provided (still verify!)
    verify = True
    ca_bundle = args.ca_bundle

    # Optional mTLS
    cert_tuple = None
    if args.mtls_cert and args.mtls_key:
        cert_tuple = (args.mtls_cert, args.mtls_key)

    token = get_token(cfg, verify, ca_bundle, args.timeout)
    s = session_with_retries()

    results = []
    for ep in cfg.get("endpoints", []):
        name = ep["name"]
        method = ep["method"].upper()
        url = f"{base}{ep['path']}"
        params = ep.get("params") or {}
        json_body = ep.get("json", None)
        expected = ep.get("expected_status", 200)
        need_keys = ep.get("expect_json_keys", [])

        headers = dict(default_headers)
        if token:
            headers["Authorization"] = f"Bearer {token}"

        t0 = time.time()
        try:
            resp = s.request(
                method, url,
                params=params,
                json=json_body,
                headers=headers,
                timeout=args.timeout,
                verify=ca_bundle if ca_bundle else verify,
                cert=cert_tuple
            )
            latency_ms = round((time.time() - t0) * 1000, 1)
            ok_status = (resp.status_code == expected)
            detail = f"status={resp.status_code}, latency_ms={latency_ms}"

            ok_json = True
            if need_keys:
                try:
                    obj = resp.json()
                    ok_json, msg = check_json_keys(obj, need_keys)
                    if not ok_json:
                        detail += f", json_check={msg}"
                except Exception as e:
                    ok_json = False
                    detail += f", json_parse_error={e}"

            ok = ok_status and ok_json
            results.append((name, ok, detail))
            print(f"[{'PASS' if ok else 'FAIL'}] {name} -> {detail}")
        except requests.exceptions.SSLError as e:
            print(f"[FAIL] {name} -> TLS/SSL error: {e}")
            results.append((name, False, f"TLS error: {e}"))
        except Exception as e:
            print(f"[FAIL] {name} -> {e}")
            results.append((name, False, str(e)))

    # Exit non-zero on any failure
    if any(not ok for _, ok, _ in results):
        sys.exit(2)

if __name__ == "__main__":
    main()

	•	Rotate tokens and certs; avoid storing production creds locally.

	•	In Kubernetes/OpenShift, env vars are set via env: in the Pod spec, often backed by ConfigMaps or Secrets.

Refactor Examples

Hardcoded:

self.accounturl = "https://api.example.com/account"
self.modelurl = "http://localhost:8000/model"

Refactored:

self.accounturl = os.getenv("ACCOUNTURL", "https://api.example.com/account")
self.modelurl = os.getenv("MODELURL", "http://localhost:8000/model")

Testing Steps

# Set and run
export ACCOUNTURL="https://test.example.com/account"
python myscript.py

# Unset → falls back to default
unset ACCOUNTURL
python myscript.py

In Kubernetes:

env:
  - name: ACCOUNTURL
    value: "https://cluster.example.com/account"

Best Practices
	•	Use .env locally (python-dotenv can help).
	•	Document required env vars in README.
	•	For sensitive values (tokens, passwords), always use Secrets, not ConfigMaps.
	•	Avoid empty string defaults unless that’s safe.
